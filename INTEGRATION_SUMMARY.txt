================================================================================
                            STORE STRUCTURE
================================================================================

useSettingsStore contains:

PROFILE:
  fullName: string
  bio: string
  profilePictureUrl: string | null
  setProfile(name, bio, url)

APPEARANCE:
  theme: 'light' | 'dark' | 'system'
  accentColor: 'blue' | 'purple' | 'green' | ... (10 total)
  setTheme(theme)
  setAccentColor(color)

NOTIFICATIONS:
  notificationsEnabled: boolean
  emailNotifications: boolean
  browserNotifications: boolean
  inAppNotifications: boolean
  pushNotifications: boolean
  habitReminders: boolean
  taskReminders: boolean
  meetingReminders: boolean
  reminderTime: string (HH:MM format)
  setNotificationSettings(settings)

EMAIL:
  emailHabitReminders: boolean
  emailTaskReminders: boolean
  emailMeetingReminders: boolean
  dailySummary: boolean
  dailySummaryTime: string (HH:MM format)
  setEmailSettings(settings)

SECURITY:
  twoFactorEnabled: boolean
  sessionTimeout: number (minutes)
  lastLogin: datetime | null
  lastPasswordChange: datetime | null
  setSecuritySettings(settings)

PREFERENCES:
  language: string (en, es, fr, de, it, pt, ja, zh, ko)
  timezone: string (UTC, EST, CST, MST, PST, GMT, IST, JST, AEST)
  setLanguage(lang)
  setTimezone(tz)

UI STATE:
  sidebarMode: 'expanded' | 'collapsed'
  setSidebarMode(mode)
  neonGlow: boolean
  setNeonGlow(bool)
  backgroundAnimation: boolean
  setBackgroundAnimation(bool)

BULK UPDATES:
  updateSettings(updates)      // Update any fields
  updateFromAPI(apiSettings)   // Sync from API
  getAllSettings()             // Get all settings

ALL PERSISTED TO localStorage & synchronized with DB

================================================================================
                           HOW TO USE IT
================================================================================

1. ACCESS SETTINGS IN ANY COMPONENT:

   import { useSettingsStore } from '../store/settingsStore';
   
   function MyComponent() {
     const { theme, accentColor, language } = useSettingsStore();
     return <div>Current theme: {theme}</div>;
   }

2. UPDATE SETTINGS:

   const { setTheme, setLanguage } = useSettingsStore();
   
   setTheme('light');       // Changes theme
   setLanguage('es');       // Changes language

3. BATCH UPDATE:

   const { updateSettings } = useSettingsStore();
   
   updateSettings({
     theme: 'dark',
     language: 'fr',
     dailySummary: false
   });

4. LISTEN TO CHANGES:

   const subscribe = useSettingsStore.subscribe(
     (state) => state.theme,
     (theme) => console.log('Theme changed to:', theme)
   );

5. GET ALL SETTINGS:

   const settings = useSettingsStore((state) => state.getAllSettings());
   console.log(settings);
